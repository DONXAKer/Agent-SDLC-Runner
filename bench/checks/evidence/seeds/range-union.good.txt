export const protocolVersion = 1;
export function mergeRanges(xs) { if (!Array.isArray(xs) || xs.some(x => !Array.isArray(x) || x.length !== 2 || !x.every(Number.isFinite) || x[0]>x[1])) throw new TypeError(); const out=[]; for (const [a,b] of xs.map(x=>[...x]).sort((x,y)=>x[0]-y[0])) { const last=out.at(-1); if(last && a<=last[1]) last[1]=Math.max(last[1],b); else out.push([a,b]); } return out; }
